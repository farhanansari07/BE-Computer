import java.util.*;

class Item {
    int weight, profit;
    double ratio;

    Item(int weight, int profit) {
        this.weight = weight;
        this.profit = profit;
        this.ratio = (double) profit / weight;
    }
}

public class FractionalKnapsack {

    static void knapsack(Item items[], int capacity) {

        // Sort items by profit/weight ratio
        Arrays.sort(items, (a, b) ->
            Double.compare(b.ratio, a.ratio)
        );

        double totalProfit = 0;

        for (int i = 0; i < items.length; i++) {

            if (capacity == 0)
                break;

            // Take complete item
            if (items[i].weight <= capacity) {

                capacity -= items[i].weight;
                totalProfit += items[i].profit;
            }

            // Take fractional item
            else {

                totalProfit += items[i].ratio * capacity;
                capacity = 0;
            }
        }

        System.out.println("Maximum Profit: " + totalProfit);
    }

    public static void main(String[] args) {

        Scanner sc = new Scanner(System.in);

        System.out.print("Enter number of items: ");
        int n = sc.nextInt();

        Item items[] = new Item[n];

        for (int i = 0; i < n; i++) {

            System.out.print("Enter weight of item " + (i+1) + ": ");
            int weight = sc.nextInt();

            System.out.print("Enter profit of item " + (i+1) + ": ");
            int profit = sc.nextInt();

            items[i] = new Item(weight, profit);
        }

        System.out.print("Enter knapsack capacity: ");
        int capacity = sc.nextInt();

        knapsack(items, capacity);

        sc.close();
    }
}

// Function to print the DP table
void printTable(const vector<vector<int>>& dp, int n, int W) {
    cout << "\nDP Table:\n";
    for (int i = 0; i <= n; i++) {
        for (int w = 0; w <= W; w++) {
            cout << setw(5) << dp[i][w];
        }
        cout << endl;
    }
}

// Function to solve the Knapsack problem using dynamic programming and return selected items
int knapsackDP(const vector<int>& weights, const vector<int>& values, int W, vector<int>& selectedItems) {
    int n = weights.size();
    vector<vector<int>> dp(n + 1, vector<int>(W + 1, 0));

    // Fill the DP table
    for (int i = 1; i <= n; i++) {
        for (int w = 1; w <= W; w++) {
            if (weights[i - 1] <= w) {
                dp[i][w] = max(dp[i - 1][w], dp[i - 1][w - weights[i - 1]] + values[i - 1]);
            } else {
                dp[i][w] = dp[i - 1][w];
            }
        }
    }

    // Print the DP table
    printTable(dp, n, W);

    // Backtrack to find the selected items
    int maxProfit = dp[n][W];
    int w = W;
    for (int i = n; i > 0 && maxProfit > 0; i--) {
        if (maxProfit != dp[i - 1][w]) {  // Item i-1 was included
            selectedItems.push_back(i);    // Save the index (1-based) of selected item
            maxProfit -= values[i - 1];
            w -= weights[i - 1];
        }
    }

    return dp[n][W];
}

 1. Aim
To write a Java program to solve the Fractional Knapsack Problem using the Greedy Method and analyze its time and space complexity.

 2. Theory
The Knapsack Problem is an optimization problem in which we are given a bag (knapsack) with a fixed capacity and a set of items, each having a weight and a profit.

The objective is to select items in such a way that the total profit is maximum without exceeding the capacity of the knapsack.

There are two main types of Knapsack Problems:
1. 0/1 Knapsack: We can either take the complete item or leave it. We cannot divide an item.
2. Fractional Knapsack: We can take the complete item or a fraction of it.
The Fractional Knapsack Problem is solved using the Greedy Method.

In this method:
- Calculate the profit-to-weight ratio of every item.
- Sort items in descending order of their ratio.
- Select the item with the highest ratio first.
- If the complete item cannot fit, take only the required fraction.
- Continue until the knapsack becomes full.
Formula:
Ratio = Profit/Weight

Maximum Profit = sum (Profit * Fraction)

 3. Real-Life Examples

1. Packing a Suitcase
Suppose you have limited luggage capacity. You select items based on their value and weight to maximize the value you can carry.

2. Gold Transportation
If you have limited carrying capacity, you can select portions of gold based on their value and weight.


3. Cargo Loading
A truck has limited capacity, so goods can be selected according to their value per unit weight.


 4. Algorithm
1. Start.
2. Read the number of items.
3. Read the weight and profit of each item.
4. Calculate the profit-to-weight ratio for every item.
5. Sort the items in descending order of their ratio.
6. Read the knapsack capacity.
7. Initialize total profit = 0.
8. Select items having the highest ratio first.
9. If the complete item fits, take it.
10. Otherwise, take the fraction that fits.
11. Calculate the total profit.
12. Print the maximum profit.
13. Stop.

Explanation:
- Time Complexity – O(n log n): Sorting takes O(n log n), and selecting items takes O(n).
- Space Complexity – O(n): We store n items in an array. Java's object-array sorting may also use additional temporary space.
